cmake_minimum_required(VERSION 3.8)
project(go8010_ros2_driver)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(robot_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
add_executable(go8010_driver_node src/go8010_driver_node.cpp)
target_include_directories(go8010_driver_node PRIVATE include)
ament_target_dependencies(go8010_driver_node rclcpp robot_msgs diagnostic_msgs)
install(TARGETS go8010_driver_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
